# Device-Driven Authorization Application Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the user-selected authorization scene with device-fact-driven request actions, add retired-product guidance, calculate approval decisions, and hard-block incompatible multi-device applications in the V2 prototype.

**Architecture:** Keep `测试设备授权平台V2.html` as the browser entry and DOM adapter. Add one UMD rules module, `authorization-application-rules.js`, for product lifecycle, device facts, request-action mapping, approval decisions, compatibility checks, and submission snapshots; the same module is loaded by Node tests and the browser. Use `node:test` plus JSDOM for pure rules and DOM behavior, while preserving the existing HCI/SCP fixed-ID and info-file behavior.

**Tech Stack:** Static HTML/CSS/JavaScript, UMD/CommonJS, Node.js 24, `node:test`, `assert`, JSDOM 26.

---

## File Structure

| File | Responsibility |
|------|----------------|
| `authorization-application-rules.js` | Pure product lifecycle, device fact, request action, approval, compatibility, and payload rules; no DOM access |
| `测试设备授权平台V2.html` | Markup, styling, Chip adapter, rendering, user interactions, mock query adapter, and submit alert |
| `tests/helpers/load-v2-dom.cjs` | Creates an isolated JSDOM instance, injects the rules module, captures alerts, and reports script errors |
| `tests/fixtures/authorization-fixtures.cjs` | Minimal trusted device facts used to cover normal, sales, no-record, and conflict cases |
| `tests/product-line-retirement.test.cjs` | Retired-product grouping, warning, selectability, and lifecycle snapshot |
| `tests/device-action-flow.test.cjs` | Device lookup states, fact summary, eligible actions, hidden compatibility scene, and reset behavior |
| `tests/approval-preview.test.cjs` | Approval boundary rules, preview invalidation, route reasons, and rule snapshots |
| `tests/multi-device-compatibility.test.cjs` | Same-context batching, hard conflict blocking, device removal recovery, and final payload |
| `tests/basic-form-layout.test.cjs` | Existing layout contract updated from visible `authScene` to device-driven `requestAction` |
| `tests/chip-hci-linkage.test.cjs` | Existing Chip and HCI/SCP regression contract |

Do not track `测试授权申请审批规则与审批流.xlsx`; the confirmed subset needed by the prototype is encoded as versioned rule data in `authorization-application-rules.js`.

```javascript
'use strict';

const fs = require('fs');
const path = require('path');
const { JSDOM, VirtualConsole } = require('jsdom');

const workspaceRoot = path.join(__dirname, '..', '..');
const htmlPath = path.join(workspaceRoot, '测试设备授权平台V2.html');
const rulesPath = path.join(workspaceRoot, 'authorization-application-rules.js');

function loadRules() {
  if (!fs.existsSync(rulesPath)) return {};
  delete require.cache[require.resolve(rulesPath)];
  return require(rulesPath);
}

function loadV2Dom(options = {}) {
  const errors = [];
  const alerts = [];
  const virtualConsole = new VirtualConsole();
  virtualConsole.on('jsdomError', (error) => errors.push(error.message));

  const dom = new JSDOM(fs.readFileSync(htmlPath, 'utf8'), {
    runScripts: options.runScripts === false ? undefined : 'dangerously',
    url: 'http://localhost/',
    virtualConsole,
    beforeParse(window) {
      window.tailwind = { config: {} };
      window.AuthorizationApplicationRules = loadRules();
      window.alert = (message) => alerts.push(String(message));
      if (options.beforeParse) options.beforeParse(window);
    }
  });

  return { dom, document: dom.window.document, window: dom.window, errors, alerts };
}

function fireChange(element, window) {
  element.dispatchEvent(new window.Event('change', { bubbles: true }));
}

module.exports = { loadV2Dom, fireChange };
```

Add to the rules module:

```javascript
function normalizeLookupResult(deviceIdentifier, result) {
  if (result && result.status === 'error') {
    return { deviceIdentifier: deviceIdentifier, lookupStatus: 'error', errorCode: result.errorCode || 'LOOKUP_FAILED' };
  }
  if (!result || result.status === 'not_found') {
    return { deviceIdentifier: deviceIdentifier, lookupStatus: 'not_found', manualReviewRequired: true };
  }
  return Object.assign({}, result.fact, { deviceIdentifier: deviceIdentifier, lookupStatus: 'success' });
}

function getLookupBlockingReason(selectedProductLineId, facts) {
  var error = facts.find(function (fact) { return fact.lookupStatus === 'error'; });
  if (error) return { code: 'LOOKUP_ERROR', deviceIdentifier: error.deviceIdentifier };
  var mismatch = facts.find(function (fact) {
    return fact.lookupStatus === 'success' && fact.productLineId !== String(selectedProductLineId);
  });
  return mismatch ? { code: 'PRODUCT_MISMATCH', deviceIdentifier: mismatch.deviceIdentifier, actual: mismatch.productLineName } : null;
}
```

```javascript
function calculateApprovalDecision(input) {
  var reasons = [];
  var decisionType = 'default_pass';
  var routeKey = 'AUTO_PASS';

  if (input.productStatus === 'retired') reasons.push('PRODUCT_RETIRED_WARNING');
  if (input.fact.selfServiceAllowed === false) {
    decisionType = 'not_allowed';
    routeKey = 'SELF_SERVICE_BLOCKED';
    reasons.push('SELF_SERVICE_NOT_AVAILABLE');
  } else if (input.fact.specialFlowKey) {
    decisionType = 'special_flow';
    routeKey = input.fact.specialFlowKey;
    reasons.push('SPECIAL_PRODUCT_FLOW');
  } else if (input.manualReviewRequired) {
    decisionType = 'manual_approval';
    routeKey = 'MANUAL_DEVICE_VERIFICATION';
    reasons.push('DEVICE_NOT_FOUND');
  } else if (input.fact.deviceSource === 'sales' && input.fact.productLineId === '20' && input.requestAction === 'add_module') {
    decisionType = 'manual_approval';
    routeKey = 'REGION_AND_HQ_MARKETING';
    reasons.push('ATRUST_SALES_ADD_MODULE');
  } else if ((input.fact.productLineId === '45' || input.fact.productLineId === '19') && Number(input.targetCapacity) > 20) {
    decisionType = 'manual_approval';
    routeKey = 'PRODUCT_LIMIT_APPROVAL';
    reasons.push('HCI_CAPACITY_OVER_20');
  } else {
    var maxApplications = input.fact.productLineId === '141' ? 1 : 2;
    if (input.requestAction === 'open' && Number(input.fact.applicationCount) >= maxApplications) {
      decisionType = 'manual_approval';
      routeKey = 'APPLICATION_LIMIT_APPROVAL';
      reasons.push('APPLICATION_COUNT_OVER_LIMIT');
    }
    var freeMonths = input.customerType === 'KA' ? 6 : 3;
    var totalMonths = Number(input.fact.testedMonths || 0) + Number(input.requestedMonths || 0);
    if (decisionType === 'default_pass' && totalMonths > freeMonths) {
      decisionType = 'manual_approval';
      routeKey = input.customerType === 'KA' ? 'KA_AND_HQ_MARKETING' : 'REGION_AND_HQ_MARKETING';
      reasons.push(input.customerType === 'KA'
        ? 'KA_DURATION_OVER_6_MONTHS'
        : 'COVERAGE_DURATION_OVER_3_MONTHS');
    }
  }

  return {
    decisionType: decisionType,
    routeKey: routeKey,
    reasonCodes: reasons,
    reasonTexts: reasons.map(function (code) { return REASON_TEXTS[code]; }),
    ruleVersion: RULE_VERSION,
    calculatedAt: input.calculatedAt || new Date().toISOString()
  };
}
```

```javascript
function findCompatibilityIssues(facts, context) {
  if (facts.length < 2) return [];
  var baseline = facts[0];
  var baselineDecision = context.decisions[0];
  var issues = [];

  facts.slice(1).forEach(function (fact, index) {
    var decision = context.decisions[index + 1];
    if (fact.productLineId !== baseline.productLineId || fact.productLineId !== String(context.productLineId)) {
      issues.push({ code: 'PRODUCT_LINE_MISMATCH', deviceIdentifier: fact.deviceIdentifier });
    }
    if (fact.deviceSource !== baseline.deviceSource) {
      issues.push({ code: 'DEVICE_SOURCE_MISMATCH', deviceIdentifier: fact.deviceIdentifier });
    }
    if (fact.deviceSource === 'borrowed' && fact.borrowOrderId !== baseline.borrowOrderId) {
      issues.push({ code: 'BORROW_ORDER_MISMATCH', deviceIdentifier: fact.deviceIdentifier });
    }
    if (fact.materialKeys.slice().sort().join('|') !== baseline.materialKeys.slice().sort().join('|')) {
      issues.push({ code: 'MATERIAL_REQUIREMENT_MISMATCH', deviceIdentifier: fact.deviceIdentifier });
    }
    if (decision.decisionType !== baselineDecision.decisionType || decision.routeKey !== baselineDecision.routeKey) {
      issues.push({ code: 'APPROVAL_ROUTE_MISMATCH', deviceIdentifier: fact.deviceIdentifier });
    }
  });
  return issues;
}
```
