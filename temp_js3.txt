        function openSpecModal(product) {
            currentSpecProduct = product;
            specModalTitle.textContent = '修改规格 - ' + (productDisplayNames[product] || product);
            specModalBody.innerHTML = renderSpecFields(product);
            specModal.classList.add('show');

            if (swgTypeSelect) {
                swgTypeSelect.addEventListener('change', function() {
                    var val = this.value;
                    document.getElementById('spec-swg-users-field').style.display = (val === '分支版') ? 'none' : '';
                    document.getElementById('spec-swg-branch-field').style.display = (val === '用户版') ? 'none' : '';
                    document.getElementById('spec-swg-bw-field').style.display = (val === '用户版') ? 'none' : '';
                });
            }
        }

        function closeSpecModal() {
            specModal.classList.remove('show');
            currentSpecProduct = '';

        function saveSpecValues() {
            var p = currentSpecProduct;
            var v = xaasSpecValues[p];
            if (!v) return;
            if (p === 'SaaS-XDR') {
                v.logAmount = parseInt(document.getElementById('spec-log').value) || 20000;
            } else if (p === 'NGDR') {
                v.version = document.getElementById('spec-version').value;
                v.pcCount = parseInt(document.getElementById('spec-pc').value) || 50;
                v.serverCount = parseInt(document.getElementById('spec-server').value) || 50;
            } else if (p === 'SaaS-aES') {
                v.version = document.getElementById('spec-version').value;
                v.pcCount = parseInt(document.getElementById('spec-pc').value) || 50;
                v.serverCount = parseInt(document.getElementById('spec-server').value) || 50;
            } else if (p === 'SASE-XDLP' || p === 'SASE-ZTNA') {
                v.userCount = parseInt(document.getElementById('spec-users').value) || 50;
            } else if (p === 'SASE-SWG') {
                v.swgType = document.getElementById('spec-swg-type').value;
                v.userCount = parseInt(document.getElementById('spec-swg-users').value) || 50;
                v.branchCount = parseInt(document.getElementById('spec-swg-branch').value) || 3;
                v.bandwidth = parseInt(document.getElementById('spec-swg-bw').value) || 100;
            } else if (p === '钓鱼邮件防护') {
                v.mailCount = parseInt(document.getElementById('spec-mail').value) || 10000;
            } else if (p === 'SASE-GA') {
                v.bandwidth = parseInt(document.getElementById('spec-bw').value) || 5;
            }
        }

        function buildDescText(product) {
            var v = xaasSpecValues[product];
            if (!v) return '';
            if (product === 'SaaS-XDR') return '有效期30天，日志量' + v.logAmount + '万条';
            if (product === 'NGDR') return '有效期30天，授权PC数' + v.pcCount + '个，授权服务器数' + v.serverCount + '个，' + v.version;
            if (product === 'SaaS-aES') return '有效期30天，授权PC数' + v.pcCount + '个，授权服务器数' + v.serverCount + '个，' + v.version;
            if (product === 'SASE-XDLP') return '有效期30天，授权用户数' + v.userCount + '个';
            if (product === 'SASE-ZTNA') return '有效期30天，授权用户数' + v.userCount + '个';
            if (product === 'SASE-SWG') {
                var t = v.swgType;
                if (t === '用户版') return '有效期30天，用户版，用户数' + v.userCount + '个';
                if (t === '分支版') return '有效期30天，分支版，分支' + v.branchCount + '*' + v.bandwidth + 'M';
                return '有效期30天，用户+分支版，用户数' + v.userCount + '个，分支' + v.branchCount + '*' + v.bandwidth + 'M';
            }
            if (product === '钓鱼邮件防护') return '有效期30天，每日邮件数量' + v.mailCount + '封';
            if (product === 'SASE-GA') return '有效期15天，带宽' + v.bandwidth + 'M';
            return '';
        }

            var el = document.getElementById(id);
            if (!el) return;

                var val = parseInt(this.value) || 0;
                if (multiplier) val = Math.round(val / multiplier);
                if (xaasSpecValues[product]) xaasSpecValues[product][fieldKey] = val;
                var item = document.querySelector('.xaas-product-item[data-product="' + product + '"]');
                if (item) {
                    var desc = item.querySelector('.xaas-product-desc');
                    if (desc) desc.textContent = buildDescText(product) || item.dataset.defaultDesc;
                }
            });
